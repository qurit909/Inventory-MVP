구글 시트(Google Sheets) 양방향 자동 연동

연동이 안 될 때 3가지 필수 체크리스트 (점검해 보세요)

  • URL 주소 확인: 주소 끝이 /edit가 아닌 /exec 로 끝나야 합니다. (자동 교정 지원)
  • 액세스 권한: [새 배포] 설정 시 '액세스 권한이 있는 사용자'를 반드시 [모든 사용자 (Anyone)]로 설정하셔야 합니다.
  • 코드 업데이트: 아래 최신 doGet + doPost 코드로 교체 후 [배포] ➔ [새 배포]를 해주세요.

배포 URL을 입력 후 [설정 저장 및 연결]을 누르면 브라우저 메모리(LocalStorage)에 안전하게 영구 저장되어 매번 새로 입력할 필요가 없습니다.

2. 구글 시트 앱 스크립트 통합 코드 (읽기 + 쓰기 지원)

구글 시트 [확장 프로그램] ➔ [Apps Script] 기존 코드를 지우고 아래 최신 코드로 덮어씌워 주세요.

// 1. 구글 시트 데이터 읽기 (대시보드로 가져오기)
function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    var result = [];
    if (data.length > 1) {
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if (!row[0] && !row[1]) continue;
        result.push({
          id: String(row[0] || ''),
          name: String(row[1] || ''),
          code: String(row[2] || ''),
          location: String(row[3] || ''),
          quantity: Number(row[4]) || 0,
          inboundDate: row[5] ? formatDate(row[5]) : '',
          soldDate: row[6] ? formatDate(row[6]) : '-',
          notes: String(row[7] || '')
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ result: "success", data: result }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ result: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function formatDate(d) {
  if (d instanceof Date) {
    var y = d.getFullYear();
    var m = ("0" + (d.getMonth() + 1)).slice(-2);
    var day = ("0" + d.getDate()).slice(-2);
    return y + "-" + m + "-" + day;
  }
  return String(d);
}

// 2. 구글 시트 데이터 쓰기 및 수정 (대시보드에서 전송)
function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({
        result: "notice", 
        message: "웹 대시보드에서 보낸 데이터가 없습니다."
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 첫 줄 헤더 자동 생성
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["제품ID", "제품명", "제품코드", "위치", "잔여수량", "입고일", "최근판매일", "메모"]);
    }
    
    // 기존 데이터 존재 시 행 수정, 없으면 새 행 추가
    var allRows = sheet.getDataRange().getValues();
    var rowIndexToUpdate = -1;
    for (var i = 1; i < allRows.length; i++) {
      if (allRows[i][0] && String(allRows[i][0]) === String(data.id)) {
        rowIndexToUpdate = i + 1;
        break;
      }
    }

    var rowValues = [
      data.id || '', 
      data.name || '', 
      data.code || '', 
      data.location || '', 
      data.quantity !== undefined ? data.quantity : 0, 
      data.inboundDate || '', 
      data.soldDate || '-', 
      data.notes || ''
    ];

    if (rowIndexToUpdate > 0) {
      sheet.getRange(rowIndexToUpdate, 1, 1, rowValues.length).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
