구글 앱스 스크립트(Apps Script) 연동 4단계 순서

  1. 사용하시는 구글 스프레드시트 개설 ➔ [확장 프로그램] ➔ [Apps Script] 클릭
  2. 기존 코드를 모두 지우고 아래 [앱스 스크립트 코드 복사] 버튼을 눌러 붙여넣기 후 저장(Ctrl+S)
  3. 우측 상단 [배포] ➔ [새 배포] ➔ 유형 선택: 웹 앱(Web App) 설정
  4. 액세스 권한을 반드시 "모든 사용자(Anyone)"로 설정 후 배포된 /exec URL 주소를 아래에 입력!
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[0] && !row[1]) continue;
    result.push({
      id: String(row[0] || ''),
      name: String(row[1] || ''),
      code: String(row[2] || ''),
      location: String(row[3] || 'A1'),
      quantity: Number(row[4] || 0),
      inboundDate: row[5] ? formatDate(row[5]) : '',
      soldDate: row[6] ? formatDate(row[6]) : '',
      notes: String(row[7] || '')
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var contents = JSON.parse(e.postData.contents);
    sheet.clear();
    sheet.appendRow(['ID', '제품명', '제품코드', '보관위치', '수량', '입고일자', '최근출고일', '메모']);
    for (var i = 0; i < contents.length; i++) {
      var item = contents[i];
      sheet.appendRow([
        item.id || '',
        item.name || '',
        item.code || '',
        item.location || '',
        item.quantity || 0,
        item.inboundDate || '',
        item.soldDate || '',
        item.notes || ''
      ]);
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'success'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function formatDate(d) {
  if (d instanceof Date) {
    return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  return String(d);
}
